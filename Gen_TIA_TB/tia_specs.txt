# =============================================================================
# tia_specs.txt - configuration for tia_tb_gen.py
#
# Every value the generator needs comes from this file. Edit the [testbench],
# [pins] and [variables] sections for your TIA, then run (from the directory
# where you start Virtuoso):
#
#     python3 tia_tb_gen.py tia_specs.txt --inspect   # check pins/devices/variables
#     python3 tia_tb_gen.py tia_specs.txt             # build the testbench
#
# Syntax: "key = value", '#' starts a comment. Numbers accept SI suffixes
# (f p n u m k M G T; note M = mega, m = milli).
# Spec syntax used below:
#     none                  no spec
#     gt VALUE | lt VALUE   output > VALUE | output < VALUE
#     range MIN MAX         MIN <= output <= MAX
#     tol TARGET PERCENT    TARGET +/- PERCENT %
# =============================================================================

[pins]
# Which pin of the TIA symbol is which. Feedback is connected
# neg_out -> pos_in and pos_out -> neg_in (pos_in -> neg_out is inverting).
# Single-ended TIA: neg_in = current input, pos_out = output,
#                   pos_in = reference input (or none), neg_out = none.

VSS     = VSS
pos_in  = INP
neg_in  = INN
pos_out = OUTP

[extra_pins]
# Every other pin of the TIA must be connected here, one line per pin:
#   PIN = VDD | VSS | gnd          tie to a supply / global ground
#   PIN = net NAME                 connect to testbench net NAME
#   PIN = vdc VALUE                voltage source VALUE to VSS
#   PIN = idc_in VALUE             DC current flowing into the pin
#   PIN = idc_out VALUE            DC current flowing out of the pin
#   PIN = noconn                   leave open
# Examples:
# VCM_REF = vdc avdd/2
# IBIAS   = idc_in 10u
# EN      = VDD
# TRIM<3:0> = VSS

# Per-transistor outputs. Every MOSFET in the TIA hierarchy gets one output per
# enabled [mos_metric:NAME] section, named NAME_<instance path>.
# =============================================================================
[mos]
tests       = OL_DC_AC        # OL_DC_AC and/or FB_AC
op_function = OS              # OS or OP, replaces {op} in the expressions
max_depth   = 12              # hierarchy levels searched below the TIA
# A transistor is any leaf instance with terminals D G S [B]. To use cell
# names instead, or to exclude cells, give case-insensitive regexes:
# cell_regex    =
# exclude_regex =

# Metric sections:
#   expression  Calculator expression; {device} = /I0/<path>, {op} = op_function
#   spec        default spec for every transistor
#   spec.GLOB   spec for matching transistors only (path below the TIA, e.g.
#               spec.M1 or spec.X2/* ; the last matching line wins);
#               "skip" = no output for those transistors
#   plot        yes/no ; enabled = yes/no

[mos_metric:region]
enabled    = yes
expression = {op}("{device}" "region")
spec       = tol 2 0.1        # 2 = saturation
plot       = yes

[mos_metric:vsat_marg]
# Saturation margin |Vds| - |Vdsat|, positive = saturated for NMOS and PMOS.
# (The raw op parameter "vsat_marg" = Vds - Vdsat is negative for a saturated
#  PMOS; use expression = {op}("{device}" "vsat_marg") if you want it anyway.)
enabled    = yes
expression = (abs({op}("{device}" "vds")) - abs({op}("{device}" "vdsat")))
spec       = gt 50m
plot       = yes
# spec.M_TAIL = gt 100m
